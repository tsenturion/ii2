## 1. Что мы будем проверять

Сейчас я создам небольшой модуль обработки заявок на возврат товара. Внутри него будет чистая бизнес-логика расчёта суммы, сервис создания возврата и простой repository, работающий с SQLite.

Я специально сделаю так, чтобы в исходной реализации находился один пограничный дефект. Hermes не будет заранее знать, где именно находится ошибка. Сначала агент должен изучить контракт, затем сгенерировать тесты на основании требований и реально запустить их.

Если тест упадёт, мы не будем автоматически переписывать сам тест только ради зелёного результата. Сначала агент должен установить, противоречит ли тест требованиям или проблема находится в production-коде.

После исправления мы отдельно измерим statement и branch coverage. Затем намеренно изменим работающий код и проверим, способны ли созданные тесты обнаружить это изменение.

```python
from __future__ import annotations

import sqlite3
from datetime import date
from decimal import Decimal


def calculate_refund(
    items: list[dict],
    requested: dict[int, int],
) -> Decimal:
    """Рассчитывает сумму возврата по выбранным позициям."""

    if not requested:
        raise ValueError("Не выбраны позиции для возврата")

    items_by_product = {item["product_id"]: item for item in items}

    total = Decimal("0")

    for product_id, requested_quantity in requested.items():
        if product_id not in items_by_product:
            raise ValueError("Товар отсутствует в заказе")

        if requested_quantity <= 0:
            raise ValueError("Количество должно быть больше нуля")

        item = items_by_product[product_id]

        if requested_quantity > item["quantity"]:
            raise ValueError("Нельзя вернуть больше товара, чем было куплено")

        unit_price = Decimal(str(item["unit_price"]))

        discount_percent = Decimal(
            str(
                item.get(
                    "discount_percent",
                    0,
                )
            )
        )

        paid_unit_price = (
            unit_price * (Decimal("100") - discount_percent) / Decimal("100")
        )

        total += paid_unit_price * requested_quantity

    return total.quantize(Decimal("0.01"))


class ReturnRepository:
    """Хранилище заявок на возврат."""

    def __init__(
        self,
        connection: sqlite3.Connection,
    ) -> None:
        self.connection = connection

        self.connection.execute(
            """
            CREATE TABLE IF NOT EXISTS returns (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                order_id INTEGER NOT NULL,
                amount TEXT NOT NULL,
                status TEXT NOT NULL
            )
            """
        )

        self.connection.commit()

    def create(
        self,
        order_id: int,
        amount: Decimal,
    ) -> dict:
        cursor = self.connection.execute(
            """
            INSERT INTO returns (
                order_id,
                amount,
                status
            )
            VALUES (?, ?, ?)
            """,
            (
                order_id,
                str(amount),
                "requested",
            ),
        )

        self.connection.commit()

        return self.get(cursor.lastrowid)

    def get(
        self,
        return_id: int,
    ) -> dict | None:
        cursor = self.connection.execute(
            """
            SELECT
                id,
                order_id,
                amount,
                status
            FROM returns
            WHERE id = ?
            """,
            (return_id,),
        )

        row = cursor.fetchone()

        if row is None:
            return None

        return {
            "id": row[0],
            "order_id": row[1],
            "amount": Decimal(row[2]),
            "status": row[3],
        }


class RefundService:
    """Создаёт заявку на возврат."""

    def __init__(
        self,
        repository: ReturnRepository,
    ) -> None:
        self.repository = repository

    def create_request(
        self,
        order: dict,
        requested: dict[int, int],
        today: date,
    ) -> dict:
        if order["status"] != "delivered":
            raise ValueError("Возврат доступен только для полученного заказа")

        delivered_at = order.get("delivered_at")

        if delivered_at is None:
            raise ValueError("Дата получения заказа неизвестна")

        days_since_delivery = (today - delivered_at).days

        if days_since_delivery < 0:
            raise ValueError("Дата получения не может быть в будущем")

        if days_since_delivery >= 14:
            raise ValueError("Срок возврата истёк")

        amount = calculate_refund(
            order["items"],
            requested,
        )

        return self.repository.create(
            order_id=order["id"],
            amount=amount,
        )
```

## 6. Создаём отдельный контракт

Содержимое:

```markdown
# Контракт модуля возвратов

Модуль реализует создание заявки на возврат товара.

## Расчёт суммы

Пользователь должен выбрать хотя бы одну позицию.

Количество возвращаемого товара должно быть больше нуля.

Нельзя вернуть товар, отсутствующий в заказе.

Нельзя вернуть количество, превышающее купленное.

Разрешено вернуть всё купленное количество позиции.

Сумма рассчитывается по цене товара на момент заказа.

Если для позиции сохранена скидка, возврат рассчитывается по фактически оплаченной цене.

Результат должен иметь точность до двух знаков после запятой.

## Создание заявки

Заявку можно создать только для заказа со статусом delivered.

Поле delivered_at обязательно.

Возврат разрешён не позднее чем через 14 календарных дней после delivered_at.

Следовательно, ровно на 14-й день возврат всё ещё разрешён.

Если текущая дата находится раньше delivered_at, данные считаются некорректными.

## Сохранение

После успешного создания заявка должна быть сохранена через ReturnRepository.

Начальный статус новой заявки:

requested

Сохранённая сумма должна совпадать с результатом calculate_refund.
```

Вот этот файл для нас важнее текущей реализации.

Если код и `CONTRACT.md` противоречат друг другу, ожидаемое поведение определяется контрактом.

## 19. Не просим автоматически добиться 100%

Если coverage оказался, например:

```text
92%
```

я не пишу модели:

```text
Сделай 100%.
```

Сначала мне нужно понять, что именно осталось непроверенным.

Поэтому мы дали агенту команду анализировать missing lines и missing branches.

100% line coverage можно получить даже достаточно слабым набором тестов. Сам процент не является нашей конечной целью.

## 20. Пятый промпт: закрываем осмысленный пробел

Если Hermes обнаружил непроверенную ветвь, отправляю следующий промпт:

```text
На основании предыдущего coverage report выбери только те непокрытые ветви testing_lab/refunds.py, которые соответствуют реальному поведению из CONTRACT.md.

Если важная ветвь действительно не проверяется, добавь минимальный тест.

Не добавляй бессмысленные тесты только ради увеличения процента coverage.

Production-код не изменяй.

После добавления теста снова запусти:

.\.venv\Scripts\python.exe -m pytest tests\testing_lab -v

а затем повтори branch coverage.

В конце объясни, какую конкретную ветвь теперь способен обнаруживать новый тест.
```

Так агент получает правильную цель: закрыть поведенческий пробел, а не просто увеличить цифру.

## 22. Теперь проверяем integration-тест отдельно

До этого pytest запускал всё вместе.

Теперь я хочу отдельно убедиться, что integration-тест действительно взаимодействует с repository.

В Terminal выполняю:

.\.venv\Scripts\python.exe -m pytest tests\testing_lab\test_refunds_integration.py -v

## 23. Проверяем, что integration-тест действительно проверяет сохранение

В integration-тесте я хочу увидеть не только вызов:

```python
service.create_request(...)
```

но и проверку того, что данные действительно оказались в repository.

Логика должна быть примерно такой:

```python
result = service.create_request(
    order=order,
    requested={101: 1},
    today=today,
)

stored = repository.get(result["id"])

assert stored is not None
assert stored["order_id"] == order["id"]
assert stored["amount"] == result["amount"]
assert stored["status"] == "requested"
```

Тогда тест проверяет границу между сервисом и хранилищем.

## 24. Проверяем качество unit-тестов через мутацию

Теперь мы знаем, что тесты зелёные.

Но возникает вопрос: способны ли они вообще ловить неправильные изменения?

Я провожу простейший mutation experiment вручную.

Сначала убеждаюсь, что Git чистый:

```powershell
git status --short
```

Если последние тесты были добавлены после предыдущего commit, сначала фиксирую их.

Теперь даю Hermes специальный промпт:

```text
Сейчас проводим учебное mutation testing.

Не исправляй тесты.

В testing_lab/refunds.py найди проверку:

requested_quantity > item["quantity"]

и временно измени оператор так, чтобы условие стало:

requested_quantity >= item["quantity"]

Больше ничего не изменяй.

После мутации запусти:

.\.venv\Scripts\python.exe -m pytest tests\testing_lab -v

Не исправляй падение.

Сообщи:

какой тест обнаружил мутацию;

какое корректное поведение нарушилось;

считается ли этот мутант обнаруженным тестовым набором.
```

Теперь мы сознательно портим production-код.

## 25. Что должна сделать эта мутация

До изменения код разрешал:

```text
requested_quantity == purchased_quantity
```

Это правильно, потому что контракт прямо разрешает вернуть всё купленное количество.

После замены:

```python
>
```

на:

```python
>=
```

полный возврат ошибочно запрещается.

Если ранее сгенерированный тест:

```text
возврат всего купленного количества
```

реально существует, он должен упасть.

Это хороший результат.

Мы специально сломали программу, и тест обнаружил изменение.

Такой мутант считается «убитым» тестами.

## 26. Анализируем mutation result с Hermes

Если тест действительно упал, задаю:

```text
Не изменяй код.

Объясни, почему эта мутация была обнаружена.

Покажи связь между:

правилом из CONTRACT.md;

изменённым условием;

конкретным упавшим тестом.

Ответь, что именно этот тест защищает от регрессии.
```

Теперь становится видно, что тест нужен не ради coverage.

Он защищает конкретное бизнес-правило:

```text
разрешено вернуть всё купленное количество позиции
```

## 27. Возвращаем production-код после мутации

После эксперимента не прошу AI вручную восстанавливать строку.

Использую Git как надёжную контрольную точку.

В Terminal:

```powershell
git restore testing_lab\refunds.py
```

Проверяю:

```powershell
git diff
```

Изменения production-кода должны исчезнуть.

Теперь ещё раз:

```powershell
.\.venv\Scripts\python.exe -m pytest tests\testing_lab -v
```

Все тесты снова должны пройти.

## 28. Проверяем более слабый тестовый набор

Теперь можно показать обратную ситуацию.

Если бы Hermes изначально не создал тест для полного количества, такая мутация могла бы остаться незамеченной.

Именно поэтому я не оцениваю suite только количеством функций `test_*`.

Два хорошо подобранных boundary-теста иногда ценнее десяти тестов одного и того же happy path.

## 29. Отдельный промпт для поиска слабых тестов

Теперь даю Hermes ещё один рабочий промпт:

```text
Проведи review только тестов из:

tests/testing_lab

Production-код не изменяй.

Для каждого теста определи, какое наблюдаемое поведение он защищает.

Найди тесты, которые:

дублируют друг друга;

проверяют детали реализации вместо внешнего поведения;

имеют слабые assertions;

могут пройти даже при существенной ошибке production-кода.

Не переписывай тесты автоматически.

Верни только анализ и конкретные рекомендации.
```

Этот промпт полезен после генерации.

AI здесь проверяет уже собственный или ранее созданный тестовый код как отдельный артефакт.

## 30. Проверяем, что тест не вычисляет ожидаемый результат тем же алгоритмом

Я отдельно открываю тест расчёта скидки.

Плохой вариант выглядел бы так:

```python
expected = price * (Decimal("100") - discount_percent) / Decimal("100")

assert calculate_refund(...) == expected
```

Если production и тест используют практически один алгоритм, одна и та же логическая ошибка может оказаться в обеих сторонах.

Для небольших известных примеров мне лучше видеть конкретный expected value:

```python
assert result == Decimal("90.00")
```

Тогда expected value независим от внутренней реализации функции.

## 31. Проверяем изоляцию unit-теста сервиса

В unit-тестах `RefundService` repository действительно можно заменить mock-объектом.

Например, я ожидаю что-то вроде:

```python
from unittest.mock import Mock

repository = Mock()

repository.create.return_value = {
    "id": 1,
    "order_id": 10,
    "amount": Decimal("90.00"),
    "status": "requested",
}
```

Здесь мы проверяем бизнес-логику самого сервиса.

Но integration-тест затем использует настоящий repository.

Таким образом, два типа тестов не дублируют друг друга, а проверяют разные границы.

## 32. Финальный промпт для Hermes

После всех изменений я отправляю агенту итоговую проверку:

```text
Проведи финальную проверку тестирования модуля testing_lab/refunds.py.

Ничего больше не изменяй.

Последовательно выполни:

2. Отдельно integration-тесты.

3. Branch coverage для пакета testing_lab.

После этого подготовь итоговый отчёт.

В отчёте раздели:

какое поведение проверяется unit-тестами;

какое взаимодействие проверяется integration-тестом;

какие fixtures используются;

какой фактический результат pytest;

какой фактический coverage;

какой пограничный production-дефект был обнаружен тестами;

какую намеренную мутацию тесты смогли обнаружить.

Не утверждай то, что не подтверждено фактическим запуском команд.
```

Это последний промпт текущей практики.

Здесь агент уже ничего не создаёт. Он собирает результаты реальных проверок.

## 34. Что именно сгенерировал AI

Теперь я отделяю результат агента от процесса проверки.

Hermes помог нам проанализировать контракт, сформировать сценарии, создать pytest-файлы и fixtures.

## 35. Итог практики

В этой работе я не давал Hermes запрос вида:

```text

```

Вместо этого мы отделили контракт от реализации и сначала заставили агента понять, какое поведение вообще необходимо проверять.

После этого AI сгенерировал unit-тесты для локальной бизнес-логики и integration-тест для взаимодействия сервиса с реальным SQLite repository.

Сгенерированные fixtures создали контролируемые данные и тестовую базу.

Первый запуск тестов обнаружил настоящий пограничный дефект: production-код запрещал возврат ровно на 14-й день, хотя контракт разрешал такой сценарий.

Мы исправили production-код, а не переписали тест под существующую ошибочную реализацию.

Затем branch coverage показал, какие направления выполнения программы действительно прошли тесты. Coverage здесь использовался как диагностический инструмент, а не как самоцель.

Наконец, мы намеренно изменили оператор сравнения в production-коде. Если тест полного возврата упал, значит набор тестов действительно способен обнаружить определённую регрессию.

В результате AI использовался не просто как генератор файлов `test_*.py`, а как участник полного проверяемого цикла: анализ требований, генерация тестов, фактический запуск, диагностика падения, проверка покрытия и оценка чувствительности тестового набора к неправильному изменению программы.
