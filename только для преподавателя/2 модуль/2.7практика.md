## Что мы будем делать

Сейчас я хочу показать следующий уровень работы с Hermes. До этого агент уже умел пользоваться встроенным терминалом, файловыми инструментами и нашим RAG-поиском. Но теперь мы создадим три специализированных Tool.

Первый инструмент будет получать `git diff`. Второй будет выполнять поиск по содержимому проекта. Третий будет запускать линтер Ruff.

Важно, что агент не будет получать произвольную строку командной оболочки. Каждый Tool будет иметь заранее определённое назначение, конкретные аргументы и собственную проверку входных данных.

Это и есть одна из причин создавать отдельные инструменты вместо постоянного использования универсального терминала.

```python
"""Реализация инструментов разработки."""

from __future__ import annotations

import json
import subprocess
from pathlib import Path


PLUGIN_DIR = Path(__file__).resolve().parent
PROJECT_ROOT = PLUGIN_DIR.parents[2]

PYTHON_EXECUTABLE = PROJECT_ROOT / ".venv" / "Scripts" / "python.exe"


def _json(data: dict) -> str:
    """Возвращает JSON с поддержкой русского текста."""

    return json.dumps(
        data,
        ensure_ascii=False,
    )


def _safe_relative_path(
    raw_path: str | None,
) -> tuple[Path | None, str | None]:
    """Проверяет, что путь находится внутри текущего проекта."""

    if raw_path is None:
        return None, None

    value = str(raw_path).strip()

    if not value:
        return None, None

    candidate = (PROJECT_ROOT / value).resolve()

    try:
        relative = candidate.relative_to(PROJECT_ROOT)

    except ValueError:
        return (
            None,
            "Путь выходит за пределы проекта.",
        )

    return relative, None


def git_diff(
    args: dict,
    **kwargs,
) -> str:
    """Возвращает обычный или staged Git diff."""

    del kwargs

    try:
        staged = bool(args.get("staged", False))

        relative_path, error = _safe_relative_path(args.get("path"))

        if error:
            return _json({"error": error})

        command = [
            "git",
            "diff",
        ]

        if staged:
            command.append("--cached")

        if relative_path is not None:
            command.extend(
                [
                    "--",
                    str(relative_path),
                ]
            )

        process = subprocess.run(
            command,
            cwd=str(PROJECT_ROOT),
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=30,
        )

        if process.returncode != 0:
            return _json(
                {"error": (process.stderr.strip() or "git diff завершился ошибкой.")}
            )

        output = process.stdout

        if len(output) > 20000:
            output = output[:20000] + "\n... результат обрезан ..."

        return _json(
            {
                "staged": staged,
                "path": (str(relative_path) if relative_path else "."),
                "diff": (output or "Изменений нет."),
            }
        )

    except subprocess.TimeoutExpired:
        return _json({"error": ("git diff превысил лимит времени.")})

    except Exception as error:
        return _json({"error": (f"Ошибка git_diff: {error}")})


def grep_search(
    args: dict,
    **kwargs,
) -> str:
    """Ищет совпадения через git grep."""

    del kwargs

    try:
        pattern = str(
            args.get(
                "pattern",
                "",
            )
        ).strip()

        if not pattern:
            return _json({"error": ("Параметр pattern не может быть пустым.")})

        if len(pattern) > 500:
            return _json({"error": ("Поисковый шаблон слишком длинный.")})

        relative_path, error = _safe_relative_path(args.get("path"))

        if error:
            return _json({"error": error})

        try:
            max_results = int(
                args.get(
                    "max_results",
                    30,
                )
            )

        except (
            TypeError,
            ValueError,
        ):
            return _json({"error": ("max_results должен быть целым числом.")})

        max_results = max(
            1,
            min(
                max_results,
                100,
            ),
        )

        command = [
            "git",
            "grep",
            "-n",
            "-I",
            "-e",
            pattern,
            "--",
        ]

        if relative_path is not None:
            command.append(str(relative_path))

        process = subprocess.run(
            command,
            cwd=str(PROJECT_ROOT),
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=30,
        )

        if process.returncode not in {
            0,
            1,
        }:
            return _json(
                {"error": (process.stderr.strip() or "git grep завершился ошибкой.")}
            )

        lines = process.stdout.splitlines()

        total_matches = len(lines)

        visible_lines = lines[:max_results]

        return _json(
            {
                "pattern": pattern,
                "path": (str(relative_path) if relative_path else "."),
                "total_matches": (total_matches),
                "shown_matches": (len(visible_lines)),
                "matches": (visible_lines),
            }
        )

    except subprocess.TimeoutExpired:
        return _json({"error": ("Поиск превысил лимит времени.")})

    except Exception as error:
        return _json({"error": (f"Ошибка grep_search: {error}")})


def run_linter(
    args: dict,
    **kwargs,
) -> str:
    """Запускает Ruff только в режиме проверки."""

    del kwargs

    try:
        if not PYTHON_EXECUTABLE.exists():
            return _json({"error": ("Не найден Python виртуального окружения.")})

        relative_path, error = _safe_relative_path(args.get("path", "."))

        if error:
            return _json({"error": error})

        target = str(relative_path) if relative_path is not None else "."

        command = [
            str(PYTHON_EXECUTABLE),
            "-m",
            "ruff",
            "check",
            target,
            "--output-format",
            "concise",
        ]

        process = subprocess.run(
            command,
            cwd=str(PROJECT_ROOT),
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=60,
        )

        output = process.stdout.strip() or process.stderr.strip()

        if len(output) > 20000:
            output = output[:20000] + "\n... результат обрезан ..."

        return _json(
            {
                "path": target,
                "exit_code": (process.returncode),
                "passed": (process.returncode == 0),
                "output": (output or "Ruff не обнаружил проблем."),
            }
        )

    except subprocess.TimeoutExpired:
        return _json({"error": ("Ruff превысил лимит времени.")})

    except Exception as error:
        return _json({"error": (f"Ошибка run_linter: {error}")})
```

Я обращаю внимание на несколько моментов.

Мы не используем `shell=True`. Команды передаются в `subprocess.run()` как массив аргументов. Это уменьшает риск того, что значение параметра превратится в дополнительную shell-команду.

Мы также проверяем путь и не позволяем инструменту работать за пределами корня проекта.

`run_linter` не использует `--fix`. Поэтому линтер только сообщает о проблемах.

И наконец, каждый обработчик возвращает JSON даже при ошибке. Такой формат соответствует контракту Hermes plugin tools.

```python
"""Регистрация инструментов dev-tools."""

from . import schemas
from . import tools


def register(ctx):
    """Регистрирует инструменты в Hermes."""

    ctx.register_tool(
        name="git_diff",
        toolset="dev_tools",
        schema=schemas.GIT_DIFF,
        handler=tools.git_diff,
    )

    ctx.register_tool(
        name="grep_search",
        toolset="dev_tools",
        schema=schemas.GREP_SEARCH,
        handler=tools.grep_search,
    )

    ctx.register_tool(
        name="run_linter",
        toolset="dev_tools",
        schema=schemas.RUN_LINTER,
        handler=tools.run_linter,
    )
```

На этом plugin состоит из четырёх файлов:

```text
plugin.yaml
schemas.py
tools.py
__init__.py
```

Это удобное разделение: manifest описывает plugin, схемы предназначены для LLM, handlers выполняют работу, а `register()` связывает их с Hermes. Такой же подход используется в официальном руководстве по созданию plugins.

## Что важно знать про git grep

Этот вариант поиска работает по файлам, которые отслеживаются Git.

Если я создал совершенно новый файл, но ещё никогда не добавлял его в репозиторий, `git grep` может его не увидеть.

Для нашей учебной кодовой базы это нужно учитывать.

Если какие-то файлы `demo_codebase` ещё не находятся под Git, я могу проверить:

```powershell
git status --short
```

и при необходимости сначала зафиксировать учебный baseline.

Сам инструмент при этом остаётся только читающим.

## Проверяем linter

Теперь отправляю:

```text
Используй run_linter.

Проверь файл:

demo_codebase/quality/bad_example.py

Ничего автоматически не исправляй.
```

Ожидаю вызов `run_linter`.

В результате Ruff должен сообщить о неиспользуемом импорте `os`.

Обычно мы увидим правило:

```text
F401
```

Поле:

```text
passed
```

будет иметь значение `false`.

Это не означает, что сам Tool сломался.

Наоборот, Tool успешно запустил линтер, а линтер обнаружил нарушение.

То есть нужно различать ошибку выполнения инструмента и отрицательный результат проверки.

## Исправляем lint-проблему вручную

Теперь для чистоты эксперимента я сам даю Hermes отдельную команду:

```text
Удалить неиспользуемый импорт os из:

demo_codebase/quality/bad_example.py

Других изменений не вносить.
```

После этого снова отправляю:

```text
Используй run_linter и повторно проверь:

demo_codebase/quality/bad_example.py
```

Теперь ожидаю:

```json
"passed": true
```

Это хороший пример цикла:

мы выполнили проверку;

получили диагностический результат;

изменили файл;

повторно запустили тот же Tool;

подтвердили устранение проблемы.

## Проверяем валидацию пути

Теперь специально проверим одно из ограничений handler.

Отправляю:

```text
Используй run_linter для пути:

..\..\

Только покажи результат инструмента.
```

Наш Tool не должен уходить выше:

```text
C:\ai-course\hermes-multiagent
```

Он должен вернуть сообщение:

```text
Путь выходит за пределы проекта.
```

Это важный момент.

Ограничение существует не только в текстовом промпте. Оно непосредственно реализовано в Python-handler.

## Проверяем валидацию grep

Теперь пишу:

```text
Вызови grep_search с пустым поисковым шаблоном и покажи ответ инструмента.
```

Если модель действительно передаст пустой `pattern`, handler вернёт:

```text
Параметр pattern не может быть пустым.
```

Если модель сама не отправит некорректный вызов из-за schema, это тоже хороший результат: ограничения параметров уже помогают ей сформировать правильный tool call.

## Проверяем Tool без прямого указания его имени

До этого я специально говорил Hermes, какой инструмент использовать.

Теперь проверим выбор инструмента моделью.

Создаю новую сессию.

Пишу:

```text
Проверь текущие незакоммиченные изменения файла dev_lab.py.

Мне нужно понять, что именно изменилось.

Ничего не изменяй.
```

Если описание `git_diff` составлено достаточно ясно, модель должна сама выбрать этот Tool.

Мне здесь важно показать студентам, зачем мы подробно писали `description`.

Мы не программировали условие:

```text
если пользователь написал "изменения", вызвать git_diff
```

LLM сама сопоставляет задачу с описанием доступной функции.

## Даём агенту комплексную задачу

Теперь проводим финальный эксперимент.

Я создаю новую сессию и отправляю обычный рабочий запрос без перечисления конкретных названий Tools:

```text
Проведи read-only анализ текущего состояния проекта.

Сначала определи, какие незакоммиченные изменения есть в dev_lab.py.

Затем найди места использования LegacyPaymentGateway в demo_codebase.

После этого выполни статическую проверку Python-кода в demo_codebase.

Никакие файлы не исправляй и не изменяй.

В конце подготовь краткий отчёт по результатам всех проверок.
```

Теперь я наблюдаю за tool calls.

Ожидаю три специализированных вызова:

```text
git_diff
grep_search
run_linter
```

Порядок может немного отличаться, если действия независимы, но все три должны быть использованы по назначению.

После каждого вызова результат возвращается LLM, и итоговый отчёт уже формируется на основании фактических данных инструментов.

## Почему это лучше одного универсального terminal

Сейчас я хочу отдельно показать студентам архитектурную разницу.

Если дать модели только универсальный терминал, она может сама сформировать:

```text
git diff
```

или команду поиска, или запуск Ruff.

Но вместе с этим она получает гораздо более широкую возможность выполнять другие команды.

Наш `git_diff` технически способен только читать diff.

`grep_search` способен только выполнять заранее определённый поиск.

`run_linter` способен только запустить Ruff без автоматического исправления.

Получается гораздо более узкая поверхность действий.

Это особенно полезно для агентов, которым нужен только анализ проекта.

## Смотрим Tools текущей сессии

Теперь ввожу:

```text
/tools list
```

Проверяю, что три инструмента всё ещё активны.

Hermes позволяет включать и отключать Tool непосредственно в текущей сессии через `/tools enable` и `/tools disable`. Отключение удаляет инструмент из доступного модели toolset.

Для эксперимента отключаю:

```text
/tools disable run_linter
```

В новой сессии спрашиваю:

```text
Каким доступным специализированным инструментом ты можешь проверить Python-код линтером?
```

`run_linter` больше не должен находиться среди доступных функций.

Возвращаю его:

```text
/tools enable run_linter
```

## Проверяем plugin через /plugins

Теперь ввожу:

```text
/plugins
```

Здесь я могу проверить, что `dev-tools` действительно относится к загруженным plugins.

Это отличается от `/tools list`.

`/plugins` показывает уровень расширения Hermes.

`/tools list` показывает функции, которые доступны модели внутри конкретной сессии.

Один plugin может зарегистрировать несколько Tools, как в нашем случае.

## Что получилось в результате

В этой работе мы создали один Hermes plugin `dev-tools`.

Внутри него зарегистрированы три независимых инструмента.

`git_diff` получает состояние изменений репозитория.

`grep_search` выполняет контролируемый поиск по Git-кодовой базе.

`run_linter` запускает Ruff без изменения файлов.

При этом LLM не получает произвольную функцию вроде «выполни любую команду». Она видит три конкретные схемы с определёнными параметрами.

Когда возникает задача, модель выбирает подходящий Tool, формирует структурированные аргументы и передаёт их Hermes.

Hermes запускает handler.

Handler проверяет аргументы, выполняет строго ограниченную операцию и возвращает JSON.

Результат снова попадает в рабочий контекст модели.

После этого агент решает, достаточно ли информации или требуется следующий Tool.

Именно на этом небольшом примере мы видим практическое устройство инструментального агента: модель отвечает за выбор следующего действия, а Tools определяют, какие реальные действия система вообще способна безопасно выполнить.
