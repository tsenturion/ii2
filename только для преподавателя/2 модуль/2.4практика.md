Теперь создадим программу, которая прочитает кодовую базу, разобьёт её на chunks и построит векторный индекс.

В центральном чате Hermes отправляю:


Код:

```python
from __future__ import annotations

import json
from pathlib import Path

import numpy as np
from sentence_transformers import SentenceTransformer


PROJECT_ROOT = Path(__file__).resolve().parents[1]

INDEX_DIR = PROJECT_ROOT / "rag" / "index"

MODEL_NAME = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"

ALLOWED_EXTENSIONS = {
    ".py",
    ".md",
    ".txt",
    ".json",
    ".yaml",
    ".yml",
    ".toml",
}

EXCLUDED_DIRECTORIES = {
    ".git",
    ".venv",
    ".hermes",
    "__pycache__",
    "node_modules",
    "models",
}

CHUNK_SIZE_LINES = 60
CHUNK_OVERLAP_LINES = 15


def should_index(path: Path) -> bool:
    """Определяет, нужно ли добавлять файл в индекс."""

    if not path.is_file():
        return False

    if path.suffix.lower() not in ALLOWED_EXTENSIONS:
        return False

    relative_path = path.relative_to(PROJECT_ROOT)

    if any(part in EXCLUDED_DIRECTORIES for part in relative_path.parts):
        return False

    if (
        len(relative_path.parts) >= 2
        and relative_path.parts[0] == "rag"
        and relative_path.parts[1] == "index"
    ):
        return False

    return True


def create_chunks(path: Path) -> list[dict]:
    """Разбивает текстовый файл на перекрывающиеся части."""

    text = path.read_text(
        encoding="utf-8",
        errors="ignore",
    )

    lines = text.splitlines()

    if not lines:
        return []

    chunks = []

    step = CHUNK_SIZE_LINES - CHUNK_OVERLAP_LINES

    start = 0

    while start < len(lines):
        end = min(
            start + CHUNK_SIZE_LINES,
            len(lines),
        )

        content = "\n".join(lines[start:end]).strip()

        if content:
            relative_path = str(path.relative_to(PROJECT_ROOT))

            chunks.append(
                {
                    "path": relative_path,
                    "start_line": start + 1,
                    "end_line": end,
                    "content": content,
                }
            )

        if end >= len(lines):
            break

        start += step

    return chunks


def main() -> None:
    """Строит локальный векторный индекс проекта."""

    INDEX_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )

    files = sorted(path for path in PROJECT_ROOT.rglob("*") if should_index(path))

    chunks = []

    for path in files:
        chunks.extend(create_chunks(path))

    if not chunks:
        raise RuntimeError("Не найдено ни одного фрагмента для индексации.")

    texts = []

    for chunk in chunks:
        texts.append(
            "Путь файла: "
            f"{chunk['path']}\n"
            "Строки: "
            f"{chunk['start_line']}-"
            f"{chunk['end_line']}\n\n"
            f"{chunk['content']}"
        )

    print(f"Найдено файлов: {len(files)}")

    print(f"Создано фрагментов: {len(chunks)}")

    print(f"Загружается embedding-модель: {MODEL_NAME}")

    model = SentenceTransformer(MODEL_NAME)

    vectors = model.encode(
        texts,
        batch_size=16,
        show_progress_bar=True,
        normalize_embeddings=True,
        convert_to_numpy=True,
    )

    vectors = np.asarray(
        vectors,
        dtype=np.float32,
    )

    np.save(
        INDEX_DIR / "vectors.npy",
        vectors,
    )

    with (INDEX_DIR / "chunks.json").open(
        "w",
        encoding="utf-8",
    ) as file:
        json.dump(
            chunks,
            file,
            ensure_ascii=False,
            indent=2,
        )

    metadata = {
        "model": MODEL_NAME,
        "files_count": len(files),
        "chunks_count": len(chunks),
        "vector_dimension": int(vectors.shape[1]),
        "chunk_size_lines": CHUNK_SIZE_LINES,
        "chunk_overlap_lines": (CHUNK_OVERLAP_LINES),
    }

    with (INDEX_DIR / "meta.json").open(
        "w",
        encoding="utf-8",
    ) as file:
        json.dump(
            metadata,
            file,
            ensure_ascii=False,
            indent=2,
        )

    print()
    print("Индекс успешно создан.")
    print(f"Каталог индекса: {INDEX_DIR}")


if __name__ == "__main__":
    main()
```

## Что именно делает индексатор

Перед запуском я коротко разбираю со студентами саму механику.

Программа проходит по проекту и выбирает текстовые файлы. `.venv`, `.git`, локальные модели и служебный каталог `.hermes` специально исключены.

Каждый файл разбивается на участки по 60 строк.

При этом между соседними фрагментами сохраняется перекрытие в 15 строк. Это нужно для того, чтобы смысл не обрывался строго на границе chunk.

Для каждого полученного фрагмента embedding-модель создаёт числовой вектор.

В результате мы сохраняем два основных массива данных.

В `chunks.json` находится исходный текст и информация о файлах.

В `vectors.npy` находятся соответствующие векторы.

Это и есть наш простой локальный векторный индекс.

Код:

```python
from __future__ import annotations

import argparse
import json
from pathlib import Path

import numpy as np
from sentence_transformers import SentenceTransformer


PROJECT_ROOT = Path(__file__).resolve().parents[1]

INDEX_DIR = PROJECT_ROOT / "rag" / "index"


def load_index() -> tuple[
    list[dict],
    np.ndarray,
    dict,
]:
    """Загружает метаданные, chunks и векторы."""

    chunks_path = INDEX_DIR / "chunks.json"

    vectors_path = INDEX_DIR / "vectors.npy"

    meta_path = INDEX_DIR / "meta.json"

    if not chunks_path.exists():
        raise FileNotFoundError("Индекс не найден. Сначала запустите build_index.py.")

    with chunks_path.open(
        "r",
        encoding="utf-8",
    ) as file:
        chunks = json.load(file)

    vectors = np.load(vectors_path)

    with meta_path.open(
        "r",
        encoding="utf-8",
    ) as file:
        metadata = json.load(file)

    return chunks, vectors, metadata


def search(
    query: str,
    top_k: int,
) -> dict:
    """Ищет наиболее близкие chunks."""

    chunks, vectors, metadata = load_index()

    model = SentenceTransformer(metadata["model"])

    query_vector = model.encode(
        [query],
        normalize_embeddings=True,
        convert_to_numpy=True,
    )[0]

    query_vector = np.asarray(
        query_vector,
        dtype=np.float32,
    )

    scores = vectors @ query_vector

    top_k = max(
        1,
        min(
            top_k,
            len(chunks),
            8,
        ),
    )

    indexes = np.argsort(scores)[::-1][:top_k]

    results = []

    for index in indexes:
        chunk = chunks[int(index)]

        results.append(
            {
                "score": round(
                    float(scores[index]),
                    4,
                ),
                "path": chunk["path"],
                "start_line": (chunk["start_line"]),
                "end_line": (chunk["end_line"]),
                "content": (chunk["content"][:4000]),
            }
        )

    return {
        "query": query,
        "results": results,
    }


def main() -> None:
    """CLI для проверки векторного поиска."""

    parser = argparse.ArgumentParser()

    parser.add_argument(
        "--query",
        required=True,
    )

    parser.add_argument(
        "--top-k",
        type=int,
        default=4,
    )

    arguments = parser.parse_args()

    result = search(
        query=arguments.query,
        top_k=arguments.top_k,
    )

    print(
        json.dumps(
            result,
            ensure_ascii=False,
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
```

Код:

```python
from __future__ import annotations

import json
import os
import subprocess
from pathlib import Path


PLUGIN_DIR = Path(__file__).resolve().parent

PROJECT_ROOT = PLUGIN_DIR.parents[2]

PYTHON_EXECUTABLE = PROJECT_ROOT / ".venv" / "Scripts" / "python.exe"

SEARCH_SCRIPT = PROJECT_ROOT / "rag" / "search_index.py"


def register(ctx):
    """Регистрирует инструмент RAG-поиска."""

    schema = {
        "name": "codebase_search",
        "description": (
            "Выполняет семантический поиск "
            "по локальному векторному индексу "
            "кодовой базы текущего проекта. "
            "Используй этот инструмент перед "
            "ответом на вопросы о структуре "
            "проекта, бизнес-логике, зависимостях, "
            "местах использования API и файлах, "
            "которые могут быть связаны с изменением."
        ),
        "parameters": {
            "type": "object",
            "properties": {
                "query": {
                    "type": "string",
                    "description": ("Поисковый запрос естественным языком."),
                },
                "top_k": {
                    "type": "integer",
                    "description": ("Количество возвращаемых фрагментов от 1 до 8."),
                    "minimum": 1,
                    "maximum": 8,
                },
            },
            "required": [
                "query",
            ],
        },
    }

    def handle_search(
        params,
        **kwargs,
    ):
        """Запускает поиск в отдельном Python-процессе."""

        del kwargs

        query = str(
            params.get(
                "query",
                "",
            )
        ).strip()

        top_k = int(
            params.get(
                "top_k",
                4,
            )
        )

        if not query:
            return json.dumps(
                {"error": ("Поисковый запрос пуст.")},
                ensure_ascii=False,
            )

        if not PYTHON_EXECUTABLE.exists():
            return json.dumps(
                {
                    "error": (
                        f"Не найден Python виртуального окружения: {PYTHON_EXECUTABLE}"
                    )
                },
                ensure_ascii=False,
            )

        if not SEARCH_SCRIPT.exists():
            return json.dumps(
                {"error": (f"Не найден скрипт поиска: {SEARCH_SCRIPT}")},
                ensure_ascii=False,
            )

        environment = dict(os.environ)

        environment["PYTHONUTF8"] = "1"

        command = [
            str(PYTHON_EXECUTABLE),
            str(SEARCH_SCRIPT),
            "--query",
            query,
            "--top-k",
            str(top_k),
        ]

        try:
            process = subprocess.run(
                command,
                cwd=str(PROJECT_ROOT),
                capture_output=True,
                text=True,
                encoding="utf-8",
                errors="replace",
                timeout=120,
                env=environment,
            )

        except subprocess.TimeoutExpired:
            return json.dumps(
                {"error": ("Поиск превысил лимит времени выполнения.")},
                ensure_ascii=False,
            )

        if process.returncode != 0:
            return json.dumps(
                {"error": (process.stderr.strip() or "Ошибка поиска.")},
                ensure_ascii=False,
            )

        return process.stdout.strip() or json.dumps(
            {"error": ("Поиск не вернул данные.")},
            ensure_ascii=False,
        )

    ctx.register_tool(
        name="codebase_search",
        toolset="code_rag",
        schema=schema,
        handler=handle_search,
    )
```

Здесь есть важный технический момент.

Сам Hermes plugin не загружает `sentence-transformers`. Он вызывает Python из нашего `.venv`, где библиотека уже установлена.

Это позволяет не смешивать зависимости самого Hermes и зависимости учебного RAG-проекта.

Я отдельно объясняю студентам, что RAG не заменяет обычные инструменты поиска.

семантический поиск становится значительно полезнее.

## Что происходит внутри нашего RAG по шагам

Когда пользователь задаёт Hermes вопрос о кодовой базе, агент сначала видит правило из `AGENTS.md`, которое требует использовать `codebase_search`.

Hermes вызывает наш plugin-tool.

Plugin запускает:

```text
rag/search_index.py
```

Поисковый запрос превращается в embedding.

Этот вектор сравнивается со всеми векторами chunks через скалярное произведение нормализованных векторов, что в нашем случае эквивалентно cosine similarity.

Затем выбираются chunks с наибольшим similarity score.

Их текст возвращается Hermes как результат tool call.

Модель получает найденный материал в текущем контексте и уже после этого формирует ответ.

Именно поэтому здесь есть обе части термина RAG.

`Retrieval` — поиск релевантных chunks.

`Augmented Generation` — генерация ответа с добавленным найденным контекстом.

## Почему мы не положили всю кодовую базу в AGENTS.md

Теперь хорошо видно различие между предыдущей и сегодняшней практикой.

`AGENTS.md` содержит устойчивые правила работы с проектом.

Он сообщает агенту, как вести себя и когда использовать retrieval.

Но исходный код туда не копируется.

Сам код находится в репозитории и в нашем векторном индексе.

Когда возникает конкретный вопрос, система извлекает только подходящие части.

Hermes загружает project context files в системную часть контекста, а динамические результаты инструментов появляются только в процессе выполнения конкретного запроса.

## Почему для занятия мы используем tool-based RAG

В Hermes существует и более автоматический вариант. Plugin может зарегистрировать `pre_llm_call` hook и перед каждым обращением к модели автоматически добавлять найденный контекст в текущий user turn. Официальная документация прямо называет этот механизм подходящим для RAG integrations.

Но для первого занятия я намеренно не использую автоматическую инъекцию.

Мне важно, чтобы студенты видели отдельный вызов:

```text
codebase_search
```

Тогда хорошо заметна архитектура RAG.

Сначала retrieval.

Затем найденный материал.

Затем генерация ответа.

Если спрятать retrieval внутрь автоматического hook уже на первом занятии, значительная часть процесса станет невидимой.

## Что мы получили в результате

Теперь у нас есть уже не просто AI-агент, который умеет читать файлы.

У нас появился отдельный слой знаний о кодовой базе.

Исходные файлы разбиваются на chunks.

Chunks преобразуются в embeddings.

Embeddings сохраняются в локальном векторном индексе.

По пользовательскому запросу выполняется semantic retrieval.

Hermes получает несколько наиболее релевантных chunks как рабочий контекст.

Модель формирует ответ уже с опорой на найденные материалы.

При этом мы не пытаемся держать весь проект в контекстном окне.

Именно в этом заключается основная идея RAG для кодовой базы: не передавать модели всё, что мы знаем о проекте, а находить и передавать то, что необходимо для конкретной задачи.
